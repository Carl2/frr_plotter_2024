  import json

  http = urllib3.PoolManager()


  headers = {
      "Content-Type": "application/json",

  }

  data = json.dumps({
      "model": "qwen2.5-coder:latest",
      "prompt": "What are the consequences of AI?",
      "stream": True
  })

  response = http.request(
      method="POST",
      url="http://localhost:11434/api/generate",
      headers=headers,
      body=data,
      preload_content=False
  )

  for chunk in response.stream():
    chunk_data = chunk.decode('utf-8')
    try:
        json_response = json.loads(chunk_data)
        if 'response' in json_response:
            print(json_response['response'], end='', flush=True)
    except json.JSONDecodeError:
        pass


  response.release_conn()

#+begin_src python
  from typing import TypeVar, Generic, Callable
  from copy import deepcopy

  T = TypeVar('T')
  U = TypeVar('U')

  class Maybe(Generic[T]):
      def __init__(self, args: T, is_ok: bool = True):
           "docstring"
           self.val = args
           self.is_ok:bool = is_ok



      def bind(self , fn: Callable[[T], 'Maybe[U]']) -> 'Maybe[U]':

          if self.is_ok:
              return fn(self.val)
          else:
              return self.val

      def __eq__(self, other):
          """
          Compare two Maybe instances for equality.
          """
          if isinstance(other, Maybe):
              return self.is_ok == other.is_ok and self.val == other.val
          elif isinstance(other, bool):
              return self.is_ok == other #and (self.val is None if not other else True)
          return False


      def __repr__(self):
          return f"Maybe(val={self.val}, is_ok={self.is_ok})"



  def error(Error: T):
      # make a deepcopy
      return Maybe(deepcopy(Error), False)

  def ok(val: T):
      #make a deepcopy
      return Maybe(deepcopy(val), True)



  def create(string: str) -> 'Maybe[T]' :
      if string is not None:
          return ok(string)
      else:
          return error("Empty string")


  def concat(original: str):

      def concat_fn(added: str) -> 'Maybe[T]':
          return ok(original + added)
      return concat_fn

        .bind(concat("Fisk"))
        )

      print(f"Not ok but ok")
  if ok("Something good" == True):

In the example =val= was an int and =Error= was a string ,
But we are not bound to that.
Quite often we want to use some kind of tracing facility.
For example how can we see what numbers were actually calculated , kind of a trace back.
First of all lets create a simple =TracerResult=, class.

The idea of the class is simple, it holds the =value=, and a =trace_log=.

The value is the current value of the calculation and the trace_log is a list of
how we got to the value.


#+NAME: trace-py-example
#+HEADER: :results output raw :exports both  :session maybe :wrap example
#+begin_src python
  from functools import reduce
  import copy

  class TracerResult(Maybe):
      def __init__(self, args: T, trace_log: list = None, is_ok: bool = True):
          super().__init__(args, is_ok)
          self.trace_log = trace_log if trace_log is not None else []

      def bind(self, fn: Callable[[T], 'TracerResult[U]']) -> 'TracerResult[U]':
          if self.is_ok:
              result = fn(self.val)
              new_trace_log = self.trace_log + [f"{fn.__name__}({self.val}) -> {result.val}"]
              return TracerResult(result.val, new_trace_log, result.is_ok)
          else:
              return TracerResult(self.val, self.trace_log, is_ok=False)

      def __repr__(self):
          return f"TracerResult(val={self.val}, trace_log={self.trace_log}, is_ok={self.is_ok})"


  # Example usage:
  def add(first_val: int):
      def add_fn(sec_val: int):
          return TracerResult(first_val + sec_val)
      return add_fn

  def mul(first_val: int):
      def mul_fn(sec_val: int):
          return TracerResult(first_val * sec_val)
      return mul_fn

  maybe_val = TracerResult(789) \
      .bind(add(2)) \
      .bind(mul(3)) \
      .bind(add(6)) \
      .bind(div(3))

  print(maybe_val)

  maybe_val = TracerResult(789) \
        .bind(add(2)) \
        .bind(div(0)) \
        .bind(add(6)) \
        .bind(div(3))

  print(maybe_val)
